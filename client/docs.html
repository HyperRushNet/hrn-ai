AIClient Documentatie

Een modulaire, belofte-gedreven JavaScript class voor het genereren van tekst en afbeeldingen via de Pollinations API. Ontworpen voor eenvoud en robuuste foutafhandeling.

Installatie

Kopieer de onderstaande class naar je project. Geen externe dependencies vereist.

class AIClient {
    constructor() {
        this.config = {
            apiKey: null,
            model: 'openai',
            systemPrompt: 'You are a helpful assistant.',
            type: 'text', // 'text' of 'image'
            width: 1024,
            height: 1024,
            history: []
        };
    }

    /**
     * Stelt de API sleutel in.
     * @param {string} key - Jouw Pollinations API key (sk_... of pk_...)
     */
    setApiKey(key) {
        if (key && typeof key === 'string') {
            this.config.apiKey = key;
        }
        return this;
    }

    /**
     * Stelt het model in.
     * @param {string} modelId - Bijv. 'openai', 'gemini', 'flux'
     */
    setModel(modelId) {
        if (modelId && typeof modelId === 'string') {
            this.config.model = modelId;
        }
        return this;
    }

    /**
     * Stelt de systeem instructie in (alleen voor text modus).
     * @param {string} prompt - De persona/instructie
     */
    setSystemPrompt(prompt) {
        if (prompt && typeof prompt === 'string') {
            this.config.systemPrompt = prompt;
        }
        return this;
    }

    /**
     * Stelt het type generatie in.
     * @param {string} type - 'text' of 'image'
     */
    setType(type) {
        if (type === 'text' || type === 'image') {
            this.config.type = type;
        } else {
            console.warn(`Type '${type}' niet herkend. Gebruik 'text' of 'image'.`);
        }
        return this;
    }

    /**
     * Stelt de breedte in (alleen voor image modus).
     */
    setWidth(width) {
        this.config.width = parseInt(width);
        return this;
    }

    /**
     * Stelt de hoogte in (alleen voor image modus).
     */
    setHeight(height) {
        this.config.height = parseInt(height);
        return this;
    }

    /**
     * Hoofdmethode voor generatie.
     * Bepaalt automatisch of er tekst of een afbeelding gegenereerd wordt.
     * @param {string} input - De prompt
     * @returns {Promise} - Tekst of Blob URL
     */
    async generate(input) {
        if (!input) throw new Error("Input prompt is vereist.");

        if (this.config.type === 'image') {
            return await this._handleImageGeneration(input);
        } else {
            return await this._handleTextGeneration(input);
        }
    }

    // --- Interne helpers ---

    async _handleTextGeneration(prompt) {
        // 1. History beheer
        this.config.history.push({ role: 'user', content: prompt });

        // 2. Request Body
        const body = {
            model: this.config.model,
            messages: [
                { role: 'system', content: this.config.systemPrompt },
                ...this.config.history
            ]
        };

        // 3. Headers
        const headers = { 'Content-Type': 'application/json' };
        if (this.config.apiKey) headers['Authorization'] = `Bearer ${this.config.apiKey}`;

        try {
            // 4. Fetch
            const response = await fetch('https://gen.pollinations.ai/v1/chat/completions', {
                method: 'POST',
                headers,
                body: JSON.stringify(body)
            });

            const data = await response.json();

            // 5. Foutafhandeling API
            if (!response.ok) {
                const errorMsg = data.error?.message || `Server error: ${response.status}`;
                throw new Error(errorMsg);
            }

            // 6. Resultaat
            const text = data.choices[0].message.content;
            
            // 7. Sla op in history
            this.config.history.push({ role: 'assistant', content: text });
            
            return text;

        } catch (error) {
            // Rollback history bij fout
            this.config.history.pop(); 
            throw error; // Gooi door naar gebruiker
        }
    }

    async _handleImageGeneration(prompt) {
        // 1. URL Parameters
        const params = new URLSearchParams();
        if (this.config.apiKey) params.append('key', this.config.apiKey);
        params.append('model', this.config.model);
        params.append('width', this.config.width);
        params.append('height', this.config.height);
        params.append('nologo', 'true');

        // 2. Fetch
        const url = `https://gen.pollinations.ai/image/${encodeURIComponent(prompt)}?${params.toString()}`;
        
        const response = await fetch(url);

        if (!response.ok) {
            throw new Error(`Afbeelding generatie mislukt: ${response.status}`);
        }

        // 3. Blob teruggeven als Object URL
        const blob = await response.blob();
        return URL.createObjectURL(blob);
    }
}

Core Methoden

setType(type)

Stelt in wat voor soort inhoud gegenereerd wordt. Dit bepaalt welk API endpoint wordt aangeroepen.

ParameterTypeOpties
typestring"text" (standaard), "image"
generate(prompt) ASYNC

Voert de generatie uit. Retourneert een Promise die resolve naar het resultaat.

ModusReturn TypeBeschrijving
TextstringDe tekst gegenereerd door de AI.
ImagestringEen tijdelijke Blob URL (bijv. blob:https://...) die direct in een <img> tag kan.

Configuratie

setApiKey(key)

Authenticatie optioneel voor sommige modellen, maar aanbevolen voor hogere limits.

setModel(modelId)

Stelt het model in. Gebruik tekst-modellen (openai, gemini) voor text en afbeeldings-modellen (flux) voor image.

setSystemPrompt(prompt)

Stelt de 'persona' in. Wordt alleen gebruikt in text modus.

setWidth(pixels)

Breedte van de afbeelding. Alleen voor image modus.

Foutafhandeling

De generate() methode werpt een Error als er iets misgaat. Gebruik try/catch.

try {
    const result = await ai.generate("Test prompt");
} catch (error) {
    console.error("Generatie mislukt:", error.message);
}

Voorbeelden

Tekst Generatie

const ai = new AIClient();

ai.setApiKey("sk_...")
  .setType("text")
  .setModel("openai")
  .setSystemPrompt("Je bent een behulpzame assistent.");

const tekst = await ai.generate("Wat is AI?");
console.log(tekst);

Afbeelding Generatie

const ai = new AIClient();

ai.setApiKey("sk_...")
  .setType("image") // Schakel naar image modus
  .setModel("flux")
  .setWidth(1920)
  .setHeight(1080);

const url = await ai.generate("Een futuristische stad");

// Toon in UI
const img = document.createElement('img');
img.src = url;
document.body.appendChild(img);
